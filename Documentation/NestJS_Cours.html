1.1 TypeScript

  • Types de base : string, number, boolean, any, unknown, never

  • Génériques : <T>, contraintes, utilitaires (Partial, Pick, Omit)
  • Décorateurs : @Component, décorateurs de classe, méthode, propriété, paramètre ⚠️ indispensable pour NestJS
  • Async/Await et Promises
  • Modules ES : import / export
// Exemple de décorateur personnalisé
function Log(target: any, key: string, descriptor: PropertyDescriptor) {
  const original = descriptor.value;
  descriptor.value = function (...args: any[]) {
    console.log(`Appel de ${key} avec`, args);
    return original.apply(this, args);
  };
}

class Service {
  @Log
  saluer(nom: string) {
    return `Bonjour ${nom}`;
  }
}

2.3 Modules

Un module est une classe annotée avec @Module() qui regroupe des composants liés.

import { Module } from '@nestjs/common';
import { UsersController } from './users.controller';
import { UsersService } from './users.service';

@Module({
  controllers: [UsersController],
  providers: [UsersService],
  exports: [UsersService], // rend le service disponible ailleurs
})
export class UsersModule {}

2.4 Controllers

Les contrôleurs gèrent les requêtes HTTP entrantes et retournent des réponses.

import { Controller, Get, Post, Body, Param, Delete } from '@nestjs/common';

@Controller('users')
export class UsersController {
  constructor(private readonly usersService: UsersService) {}

  @Get()
  findAll() {
    return this.usersService.findAll();
  }

  @Get(':id')
  findOne(@Param('id') id: string) {
    return this.usersService.findOne(+id);
  }

  @Post()
  create(@Body() createUserDto: CreateUserDto) {
    return this.usersService.create(createUserDto);
  }

  @Delete(':id')
  remove(@Param('id') id: string) {
    return this.usersService.remove(+id);
  }
}

2.5 Providers & Injection de dépendances

Un provider est une classe annotée @Injectable() qui peut être injectée dans d'autres classes via le constructeur.

import { Injectable } from '@nestjs/common';

@Injectable()
export class UsersService {
  private users = [];

  findAll() { return this.users; }

  findOne(id: number) {
    return this.users.find(u => u.id === id);
  }

  create(dto: CreateUserDto) {
    const user = { id: Date.now(), ...dto };
    this.users.push(user);
    return user;
  }

  remove(id: number) {
    this.users = this.users.filter(u => u.id !== id);
  }
}

3.3 Guards, Interceptors, Exception Filters

Guards — Contrôle d'accès

@Injectable()
export class AuthGuard implements CanActivate {
  constructor(private jwtService: JwtService) {}

  canActivate(context: ExecutionContext): boolean {
    const request = context.switchToHttp().getRequest();
    const token = request.headers.authorization?.split(' ')[1];
    if (!token) throw new UnauthorizedException();
    try {
      request.user = this.jwtService.verify(token);
      return true;
    } catch {
      throw new UnauthorizedException();
    }
  }
}

// Utilisation
@UseGuards(AuthGuard)
@Get('profile')
getProfile(@Req() req) {
  return req.user;
}

Interceptors — Transformation des réponses

@Injectable()
export class TransformInterceptor implements NestInterceptor {
  intercept(context: ExecutionContext, next: CallHandler): Observable<any> {
    return next.handle().pipe(
      map(data => ({ success: true, data, timestamp: new Date() })),
    );
  }
}

Exception Filters — Gestion centralisée des erreurs

@Catch(HttpException)
export class HttpExceptionFilter implements ExceptionFilter {
  catch(exception: HttpException, host: ArgumentsHost) {
    const ctx = host.switchToHttp();
    const response = ctx.getResponse();
    const status = exception.getStatus();

    response.status(status).json({
      statusCode: status,
      message: exception.message,
      timestamp: new Date().toISOString(),
    });
  }
}

3.4 Authentification JWT

npm install @nestjs/jwt @nestjs/passport passport passport-jwt bcrypt
npm install -D @types/passport-jwt @types/bcrypt
// auth.service.ts
@Injectable()
export class AuthService {
  constructor(
    private usersService: UsersService,
    private jwtService: JwtService,
  ) {}

  async register(dto: RegisterDto) {
    const hash = await bcrypt.hash(dto.password, 10);
    return this.usersService.create({ ...dto, password: hash });
  }

  async login(email: string, password: string) {
    const user = await this.usersService.findByEmail(email);
    if (!user || !(await bcrypt.compare(password, user.password))) {
      throw new UnauthorizedException('Identifiants invalides');
    }
    const payload = { sub: user.id, email: user.email };
    return { access_token: this.jwtService.sign(payload) };
  }
}

RBAC — Contrôle d'accès basé sur les rôles

export const Roles = (...roles: string[]) => SetMetadata('roles', roles);

@Injectable()
export class RolesGuard implements CanActivate {
  constructor(private reflector: Reflector) {}

  canActivate(context: ExecutionContext): boolean {
    const required = this.reflector.get<string[]>('roles', context.getHandler());
    if (!required) return true;
    const { user } = context.switchToHttp().getRequest();
    return required.includes(user.role);
  }
}

// Utilisation
@Roles('admin')
@UseGuards(AuthGuard, RolesGuard)
@Delete(':id')
remove(@Param('id') id: string) { /* ... */ }

4.1 Architecture avancée

  • Circular dependencies : forwardRef()

// Module dynamique
@Module({})
export class DatabaseModule {
  static forRoot(config: DbConfig): DynamicModule {
    return {
      module: DatabaseModule,
      providers: [
        { provide: 'DB_CONFIG', useValue: config },
        DatabaseService,
      ],
      exports: [DatabaseService],
      global: true,
    };
  }
}

4.2 Microservices

  • Transporters : TCP, Redis, RabbitMQ, Kafka, gRPC, NATS
  • Communication synchrone (send) vs asynchrone (emit)
  • Patterns : Request-Response, Event-Based
// main.ts (microservice)
const app = await NestFactory.createMicroservice<MicroserviceOptions>(AppModule, {
  transport: Transport.REDIS,
  options: { host: 'localhost', port: 6379 },

await app.listen();

// Client
@Injectable()
export class UsersClient {
  constructor(@Inject('USERS_SERVICE') private client: ClientProxy) {}

  findAll() {
    return this.client.send({ cmd: 'get_users' }, {});
  }

  async createUser(data: any) {
    this.client.emit('user_created', data).subscribe();
  }
}

4.5 Performance & Cache

npm install @nestjs/cache-manager cache-manager redis
@Injectable()
export class UsersService {
  constructor(
    @Inject(CACHE_MANAGER) private cache: Cache,
    private repo: UsersRepository,
  ) {}

  async findOne(id: number) {
    const cached = await this.cache.get(`user:${id}`);
    if (cached) return cached;

    const user = await this.repo.findOne(id);
    await this.cache.set(`user:${id}`, user, 60000);
    return user;
  }
}

Test unitaire (Jest)

describe('UsersService', () => {
  let service: UsersService;
  let repo: Repository<User>;

    const module = await Test.createTestingModule({
      providers: [
        UsersService,
        { provide: getRepositoryToken(User), useValue: mockRepo },
      ],
    }).compile();

    service = module.get(UsersService);
    repo = module.get(getRepositoryToken(User));
  });

  it('devrait créer un utilisateur', async () => {
    const dto = { email: 'test@test.com', password: 'pass' };
